<script>
const base = '';  

// ===== TRANSCRIBE & AUTO-TRANSLATE =====
async function transcribeAndTranslate() {
  const file = document.getElementById('audioFile').files[0];
  if (!file) return alert('Please select an audio file.');
  
  const model = document.getElementById('whisperOverride').value;
  const fd = new FormData();
  fd.append('file', file);
  if (model) fd.append('model', model);
  
  const transcribeResultBox = document.getElementById('transcribeResult');
  const translationSection = document.getElementById('translationSection');
  const autoTranslationBox = document.getElementById('autoTranslationResult');
  
  transcribeResultBox.textContent = '⏳ Transcribing…';
  translationSection.style.display = 'none';
  
  try {
    // Step 1: Transcribe
    const transcribeRes = await fetch(`${base}/transcribe/`, { 
      method: 'POST', 
      body: fd 
    });
    const transcribeData = await transcribeRes.json();
    
    const transcript = transcribeData.transcript;
    const detectedLanguage = transcribeData.detected_language;
    
    transcribeResultBox.textContent = `Language: ${detectedLanguage}\n\n${transcript}`;
    
    // Step 2: Auto-translate
    autoTranslationBox.textContent = '⏳ Translating…';
    translationSection.style.display = 'block';
    
    const translateRes = await fetch(`${base}/transcribe-and-translate`, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({
        transcript: transcript,
        detected_language: detectedLanguage
      })
    });
    
    const translateData = await translateRes.json();
    autoTranslationBox.textContent = `${translateData.translation}`;
    
  } catch(e) { 
    transcribeResultBox.textContent = '❌ Error: ' + e;
    translationSection.style.display = 'none';
  }
}

// ===== MANUAL TRANSLATE FUNCTION =====
async function translate() {
  const text = document.getElementById('translateText').value.trim();
  const direction = document.getElementById('direction').value;
  const resultBox = document.getElementById('translateResult');

  if (!text) {
    alert('Enter text to translate.');
    return;
  }

  resultBox.textContent = '⏳ Translating…';
  console.log('Translating:', { text, direction });

  try {
    const url = `${base}/translate`;
    console.log('Fetching from:', url);

    const response = await fetch(url, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ text, direction })
    });

    console.log('Response status:', response.status);

    if (!response.ok) {
      const errorText = await response.text();
      resultBox.textContent = `❌ Error ${response.status}: ${errorText}`;
      console.error('Server error:', errorText);
      return;
    }

    const data = await response.json();
    console.log('Translation result:', data);
    resultBox.textContent = data.translated || data.text || '(empty response)';

  } catch (error) {
    resultBox.textContent = `❌ Error: ${error.message}\n\nMake sure backend is running at ${base}`;
    console.error('Translate error:', error);
  }
}

async function fetchModels() {
  try {
    const r = await fetch(`${base}/models`);
    const d = await r.json();
    document.getElementById('modelResult').textContent = JSON.stringify(d, null, 2);
  } catch(e) { document.getElementById('modelResult').textContent = '❌ ' + e; }
}

async function updateModels() {
  const w = document.getElementById('cfgWhisper').value;
  if (!w) return alert('Please select a Whisper model to update.');
  try {
    const r = await fetch(`${base}/models`, {
      method: 'PATCH',
      headers: {'Content-Type':'application/json'},
      body: JSON.stringify({ whisper_model: w })
    });
    const d = await r.json();
    document.getElementById('modelResult').textContent = '✅ Updated:\n' + JSON.stringify(d, null, 2);
  } catch(e) { document.getElementById('modelResult').textContent = '❌ ' + e; }
}

document.getElementById('translateBtn').addEventListener('click', translate);

fetchModels();
