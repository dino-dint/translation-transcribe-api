// HELPERS

function setResult(id, text, state) {
  const box = document.getElementById(id);
  if (!box) return;
  box.style.display = 'block';
  const btn = box.querySelector('.copy-btn');
  box.className = 'result-box' + (state ? ' ' + state : '');
  if (btn) box.appendChild(btn); // re-attach so it's last
  // Clear text nodes only
  Array.from(box.childNodes).forEach(n => {
    if (n.nodeType === Node.TEXT_NODE) box.removeChild(n);
  });
  box.insertBefore(document.createTextNode(text), btn || null);
}

function setBtn(id, disabled, label) {
  const b = document.getElementById(id);
  if (!b) return;
  b.disabled = disabled;
  b.textContent = label;
}

function copyResult(id) {
  const box = document.getElementById(id);
  const btn = box.querySelector('.copy-btn');
  const text = (box.innerText || box.textContent)
    .replace(/^copy$|^copied!$/gm, '').trim();
  navigator.clipboard.writeText(text).then(() => {
    if (btn) { btn.textContent = 'copied!'; setTimeout(() => btn.textContent = 'copy', 1600); }
  });
}

//  TAB SWITCHER 

function switchTab(name, el) {
  ['status', 'whisper', 'provider'].forEach(t => {
    document.getElementById('tab-' + t).style.display = (t === name) ? 'block' : 'none';
  });
  document.querySelectorAll('.tab').forEach(b => b.classList.remove('active'));
  el.classList.add('active');
}

// FILE DROP ZONE

function onFileSelect(input) {
  const fi = document.getElementById('fileInfo');
  if (input.files[0]) {
    fi.style.display = 'flex';
    fi.innerHTML = '🎵 <strong style="color:var(--gold2)">' + input.files[0].name + '</strong>'
      + ' <span style="color:var(--muted)">(' + (input.files[0].size / 1024).toFixed(0) + ' KB)</span>';
  }
}

const dropZone = document.getElementById('dropZone');
dropZone.addEventListener('dragover', e => { e.preventDefault(); dropZone.classList.add('over'); });
dropZone.addEventListener('dragleave', () => dropZone.classList.remove('over'));
dropZone.addEventListener('drop', e => {
  e.preventDefault();
  dropZone.classList.remove('over');
  const f = e.dataTransfer.files[0];
  if (f) {
    // Assign to file input
    const dt = new DataTransfer();
    dt.items.add(f);
    document.getElementById('audioFile').files = dt.files;
    onFileSelect({ files: [f] });
  }
});

function clearTranscribe() {
  document.getElementById('audioFile').value = '';
  document.getElementById('fileInfo').style.display = 'none';
  document.getElementById('fileInfo').innerHTML = '';
  setResult('transcribeResult', 'Upload a file and click Transcribe & Translate…', '');
  setResult('autoTranslateResult', 'Translation will appear here after transcription…', '');
}

//  DIRECTION SWAP 

function swapDirection() {
  const f = document.getElementById('dirFrom');
  const t = document.getElementById('dirTo');
  const tmp = f.value;
  f.value = t.value;
  t.value = tmp;
}

// TRANSCRIBE + AUTO-TRANSLATE 

async function doTranscribeAndTranslate() {
  const file = document.getElementById('audioFile').files[0];
  if (!file) { alert('Please select an audio file first.'); return; }

  const model = document.getElementById('whisperOverride').value;

  setBtn('transcribeBtn', true, '⏳ Transcribing…');
  setResult('transcribeResult', '⏳ Transcribing audio, please wait…', 'loading');
  setResult('autoTranslateResult', 'Waiting for transcript…', 'loading');

  let transcript = '', detectedLang = '';

  // Step 1: Transcribe
  try {
    const fd = new FormData();
    fd.append('file', file);
    if (model) fd.append('model', model);

    const r = await fetch('/transcribe/', { method: 'POST', body: fd });
    const raw = await r.text();

    if (!r.ok) {
      setResult('transcribeResult', '❌ Transcription error ' + r.status + ':\n' + raw, 'error');
      setResult('autoTranslateResult', '⚠️ Cannot translate — transcription failed.', 'error');
      setBtn('transcribeBtn', false, ' Transcribe & Translate');
      return;
    }

    const d = JSON.parse(raw);
    transcript   = d.transcript;
    detectedLang = d.detected_language;

    setResult(
      'transcribeResult',
      'Language detected: ' + detectedLang.toUpperCase() + '\n\n' + transcript,
      'success'
    );
  } catch (e) {
    setResult('transcribeResult', '❌ Network error: ' + e.message, 'error');
    setResult('autoTranslateResult', '⚠️ Cannot translate — transcription failed.', 'error');
    setBtn('transcribeBtn', false, '▶ Transcribe & Translate');
    return;
  }

  // Step 2: Auto-translate
  setBtn('transcribeBtn', true, '⏳ Translating…');
  setResult('autoTranslateResult', '⏳ Translating transcript…', 'loading');

  const direction = detectedLang === 'km' ? 'km_to_en' : 'en_to_km';
  const dirLabel  = direction === 'km_to_en' ? 'Khmer → English' : 'English → Khmer';

  try {
    const r = await fetch('/translate', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ text: transcript, direction })
    });

    const raw = await r.text();

    if (!r.ok) {
      setResult('autoTranslateResult', '❌ Translation error ' + r.status + ':\n' + raw, 'error');
      return;
    }

    const d = JSON.parse(raw);
    setResult(
      'autoTranslateResult',
      '(' + dirLabel + ')\n\n' + (d.translated || '(empty)'),
      'success'
    );
  } catch (e) {
    setResult('autoTranslateResult', '❌ Network error: ' + e.message, 'error');
  } finally {
    setBtn('transcribeBtn', false, '▶ Transcribe & Translate');
  }
}

// MANUAL TRANSLATE 

async function doManualTranslate() {
  const text = document.getElementById('translateText').value.trim();
  const from = document.getElementById('dirFrom').value;
  const to   = document.getElementById('dirTo').value;

  if (!text) { alert('Please enter text to translate.'); return; }
  if (from === to) { alert('Source and target languages must be different.'); return; }

  const direction = from + '_to_' + to;

  setBtn('translateBtn', true, '⏳ Translating…');
  setResult('translateResult', '⏳ Translating…', 'loading');

  try {
    const r = await fetch('/translate', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ text, direction })
    });

    const raw = await r.text();

    if (!r.ok) {
      setResult('translateResult', '❌ Error ' + r.status + ':\n' + raw, 'error');
      return;
    }

    const d = JSON.parse(raw);
    setResult('translateResult', d.translated || '(empty response)', 'success');
  } catch (e) {
    setResult('translateResult', '❌ Network error: ' + e.message, 'error');
  } finally {
    setBtn('translateBtn', false, '▶ Translate');
  }
}

function clearManual() {
  document.getElementById('translateText').value = '';
  setResult('translateResult', 'Result will appear here…', '');
}

async function doFetchModel() {
  setResult('modelResult', '⏳ Fetching current config…', 'loading');
  try {
    // NOTE: update this path to match your actual FastAPI router prefix
    // e.g. if router has prefix="/api" → use "/api/models"
    const r = await fetch('/models');
    const d = await r.json();

    document.getElementById('sv-whisper').textContent  = d.whisper_model        || '—';
    document.getElementById('sv-provider').textContent = d.translation_provider || '—';
    document.getElementById('sv-model').textContent    = d.openai_model         || d.gemini_model || '—';

    setResult('modelResult', JSON.stringify(d, null, 2), 'success');
  } catch (e) {
    setResult('modelResult', '❌ ' + e.message, 'error');
  }
}

async function saveConfig(scope) {
  const body = {};
  let resultId = 'modelResult';

  if (scope === 'whisper') {
    resultId = 'whisperSaveResult';
    const w = document.getElementById('cfgWhisper').value;
    if (!w) { alert('Please select a Whisper model first.'); return; }
    body.whisper_model = w;
  } else {
    resultId = 'providerSaveResult';
    const p  = document.getElementById('cfgProvider').value;
    const om = document.getElementById('cfgOpenAIModel').value;
    if (!p) { alert('Please select a provider first.'); return; }
    body.translation_provider = p;
    if (om) body.openai_model = om;
  }

  setResult(resultId, '⏳ Saving…', 'loading');

  try {
    const r = await fetch('/models', {
      method: 'PATCH',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify(body)
    });
    const d = await r.json();
    // Update status cards
    document.getElementById('sv-whisper').textContent  = d.whisper_model        || '—';
    document.getElementById('sv-provider').textContent = d.translation_provider || '—';
    document.getElementById('sv-model').textContent    = d.openai_model         || d.gemini_model || '—';
    setResult(resultId, '✅ Saved:\n\n' + JSON.stringify(d, null, 2), 'success');
  } catch (e) {
    setResult(resultId, '❌ ' + e.message, 'error');
  }
}
